112    def set_triangulation(self):
113        """
114            Setting up triangulations for the FEM grid
115            U = cell array with each element = array of vertices of Upper Triangle of
116            the rectangular cell
117            L = cell array with each element = array of vertices of Lower Triangle of
118            the rectangular cell
119            At every point (i,j), U{i,j} & L{i,j} are cells with coordinates of vertices
120            of the two triangles obtained by bisecting the rectangle starting at
121            (i,j). The bisection line goes from NW to SE.
122        """
123        self.U = np.empty((self.m, self.n), dtype=object)
124        self.L = np.empty((self.m, self.n), dtype=object)
125
126        for j in range(self.m):
127            for k in range(self.n):
128                x1 = self.left + j * self.dx
129                y1 = self.bottom + k * self.dy
130                x2 = self.left + (j + 1) * self.dx
131                y2 = y1
132                x3 = x1
133                y3 = self.bottom + (k + 1) * self.dy
134                x4 = x2
135                y4 = y3
136
137                # lower triangle vertices
138                l = {"x": np.array([x1, x2, x3]), "y": np.array([y1, y2, y3])}
139
140                # upper triangle vertices
141                u = {"x": np.array([x4, x3, x2]), "y": np.array([y4, y3, y2])}
142
143                self.U[j, k] = u
144                self.L[j, k] = l

146    def _polyarea(self, x, y):
147        """
148        Calculate the area of a polygon using the Shoelace formula.
149        The vertices are defined by the x and y coordinates.
150
151        Args:
152            x (list or array): x-coordinates of the polygon vertices
153            y (list or array): y-coordinates of the polygon vertices
154
155        Returns: (float)
156            Area of the polygon
157        """
158        return 0.5 * abs(
159            sum(x[i] * y[i + 1] - y[i] * x[i + 1] for i in range(-1, len(x) - 1))
160        )

162    def _beta_func(self, x, y, beta):
163        """
164        Evaluates the coefficient beta at each grid point
165        $$ \beta = K \lambda $$
166        x and y are the coordinates of the grid point
167        The corresponding index locations in the matrix for beta
168        are determined in mm and nn respectively.
169
170        Analogous to beta_func.m in the MATLAB code
171        """
172        mm = round((x - self.left) / self.dx)
173        nn = round((y - self.bottom) / self.dy)
174
175        return beta[
176            int(nn), int(mm)
177        ]  # nn and mm are casted to integers for indexing purposes

198            FIXME: need to add definition here

200        beta_1 = self._beta_func(T["x"][0], T["y"][0], beta)

510    def _f_func(self, x, y, f):
511        """
512        Evaluates the source term 'f' at each grid point
513            % f is non-zero only at injection and production wells
514            % x and y are the coordinates of the grid point
515            % The corresponding index locations in the matrix for f
516            % are determined in mm and nn respectively.
517        """
518        mm = int(round((x - self.left) / self.dx))
519        nn = int(round((y - self.bottom) / self.dy))
520
521        return f[nn, mm]

523    def _trmatrix(self, T, f_1, f_2, f_3, v):
524        s = self._polyarea(T["x"], T["y"])
525        f_c = (f_1 + f_2 + f_3) / 3
526        v_c = (v[0] + v[1] + v[2]) / 3
527        f_4 = (f_2 + f_3) / 2
528        f_5 = (f_1 + f_3) / 2
529        f_6 = (f_1 + f_2) / 2
530
531        v_4 = (v[1] + v[2]) / 2
532        v_5 = (v[2] + v[0]) / 2
533        v_6 = (v[0] + v[1]) / 2
534
535        return (f_4 * v_4 + f_5 * v_5 + f_6 * v_6 + f_c * v_c) * s / 4
